services:
  db:
    image: supabase/postgres:15.14.1.130
    container_name: supabase-db
    restart: unless-stopped
    env_file:
      - /mnt/user/appdata/supabase/.env
    environment:
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_USER: ${POSTGRES_USER}
    volumes:
      - /mnt/user/appdata/supabase/db:/var/lib/postgresql/data
    ports:
      - "5432:5432"

  meta:
    image: supabase/postgres-meta:v0.96.6
    container_name: supabase-meta
    restart: unless-stopped
    env_file:
      - /mnt/user/appdata/supabase/.env
    environment:
      PG_META_DB_HOST: db
      PG_META_DB_PORT: "5432"
      PG_META_DB_NAME: ${POSTGRES_DB}
      PG_META_DB_USER: ${POSTGRES_USER}
      PG_META_DB_PASSWORD: ${POSTGRES_PASSWORD}
    depends_on:
      - db

  rest:
    image: postgrest/postgrest
    container_name: supabase-rest
    restart: unless-stopped
    env_file:
      - /mnt/user/appdata/supabase/.env
    environment:
      PGRST_DB_URI: postgres://${POSTGRES_USER}:${POSTGRES_PASSWORD}@db:5432/${POSTGRES_DB}
      PGRST_DB_ANON_ROLE: ${POSTGRES_USER}
      PGRST_DB_SCHEMAS: public
    depends_on:
      - db
    ports:
      - "3001:3000"

  studio:
    image: supabase/studio:2026.02.16-sha-26c615c
    container_name: supabase-studio
    restart: unless-stopped
    env_file:
      - /mnt/user/appdata/supabase/.env
    environment:
      HOSTNAME: "0.0.0.0"
      STUDIO_PG_META_URL: http://meta:8080

      POSTGRES_HOST: db
      POSTGRES_PORT: "5432"
      POSTGRES_DB: ${POSTGRES_DB}
      POSTGRES_PASSWORD: ${POSTGRES_PASSWORD}
      POSTGRES_USER_READ_WRITE: ${POSTGRES_USER}

      SNIPPETS_MANAGEMENT_FOLDER: /app/snippets
    volumes:
      - /mnt/user/appdata/supabase/snippets:/app/snippets
    depends_on:
      - meta
    ports:
      - "3000:3000"
